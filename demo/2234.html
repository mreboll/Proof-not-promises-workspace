using System.Linq;
using System.Threading;
using System.Threading.Tasks;
using Microsoft.CodeAnalysis;
using Microsoft.CodeAnalysis.CSharp;
using Microsoft.CodeAnalysis.Diagnostics;
using Microsoft.CodeAnalysis.Testing;
using MessagePack.SourceGenerator;
using MessagePack.SourceGenerator.Analyzers;
using Xunit;

public class LedgerIssue2234AnalyzerSuppressSourceGeneration
{
    [Fact]
    public async Task AnalyzerReportsDuplicateKeyWhenSourceGenerationIsSuppressed()
    {
        const string source = """
            using MessagePack;

            namespace TempProject
            {
                [MessagePackObject(SuppressSourceGeneration = true)]
                public class MyObject
                {
                    [Key(0)]
                    public int First { get; set; }

                    [Key(0)]
                    public int Second { get; set; }
                }
            }
            """;

        ImmutableArray<Diagnostic> analyzerDiagnostics = await GetAnalyzerDiagnosticsAsync(source);
        var msgPackDiagnostics = analyzerDiagnostics.Where(static d => d.Id.StartsWith("MsgPack")).ToImmutableArray();

        Assert.True(msgPackDiagnostics.Length > 0, "The analyzer did not report any MsgPack diagnostics for a type with SuppressSourceGeneration = true.");
    }

    [Fact]
    public async Task AnalyzerReportsNothingForValidTypeWhenSourceGenerationIsSuppressed()
    {
        const string source = """
            using MessagePack;

            namespace TempProject
            {
                [MessagePackObject(SuppressSourceGeneration = true)]
                public class MyObject
                {
                    [Key(0)]
                    public int First { get; set; }

                    [Key(1)]
                    public int Second { get; set; }
                }
            }
            """;

        ImmutableArray<Diagnostic> analyzerDiagnostics = await GetAnalyzerDiagnosticsAsync(source);
        var msgPackDiagnostics = analyzerDiagnostics.Where(static d => d.Id.StartsWith("MsgPack")).ToImmutableArray();

        Assert.Empty(msgPackDiagnostics);
    }

    private static async Task<ImmutableArray<Diagnostic>> GetAnalyzerDiagnosticsAsync(string source)
    {
        var parseOptions = new CSharpParseOptions(LanguageVersion.Latest);
        var syntaxTree = CSharpSyntaxTree.ParseText(source, parseOptions);

        var references = ReferencesHelper.GetReferences(ReferencesSet.MessagePack)
            .Concat(await ReferenceAssemblies.Net.Net80.ResolveAsync(LanguageNames.CSharp, CancellationToken.None));

        var compilation = CSharpCompilation.Create(
            "TestAssembly",
            new[] { syntaxTree },
            references,
            new CSharpCompilationOptions(OutputKind.DynamicallyLinkedLibrary));

        GeneratorDriver driver = CSharpGeneratorDriver.Create(new MessagePackGenerator().AsSourceGenerator());
        driver = driver.RunGeneratorsAndUpdateCompilation(compilation, out Compilation output, out var _);

        return await output
            .WithAnalyzers(ImmutableArray.Create<DiagnosticAnalyzer>(new MsgPack00xMessagePackAnalyzer()))
            .GetAnalyzerDiagnosticsAsync();
    }
}
