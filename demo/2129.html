public class LedgerIssue2129

    private readonly ITestOutputHelper testOutputHelper;

    public LedgerIssue2129(ITestOutputHelper testOutputHelper)
    {
        this.testOutputHelper = testOutputHelper;
    }

    [Fact]
    public async Task ExternalTypeFormatter_WithSurrogateType_GeneratesFormatter()
    {
        var net80Refs = await ReferenceAssemblies.Net.Net80.ResolveAsync(LanguageNames.CSharp, CancellationToken.None);

        string projectASource = """
namespace ProjectA
{
    public class ExternalType
    {
        public int Id { get; set; }
        public string Name { get; set; } = string.Empty;
    }
}
""";

        CSharpCompilation projectACompilation = CSharpCompilation.Create(
            "ProjectA",
            new[] { CSharpSyntaxTree.ParseText(projectASource, new CSharpParseOptions(LanguageVersion.Latest)) },
            net80Refs,
            new CSharpCompilationOptions(OutputKind.DynamicallyLinkedLibrary));

        using var projectAStream = new MemoryStream();
        var projectAEmitResult = projectACompilation.Emit(projectAStream);
        projectAEmitResult.Success.Should().BeTrue();
        var projectARef = MetadataReference.CreateFromImage(projectAStream.ToArray());

        string projectBSource = """
using MessagePack;
using MessagePack.Formatters;
using ProjectA;

namespace ProjectB
{
    [MessagePackObject]
    public class ExternalTypeSurrogate
    {
        [Key(0)]
        public int Id { get; set; }

        [Key(1)]
        public string Name { get; set; } = string.Empty;
    }

    [GeneratedMessagePackFormatter(SurrogateType = typeof(ExternalTypeSurrogate))]
    public partial class ExternalTypeFormatter : IMessagePackFormatter<ExternalType>
    {
    }
}
""";

        var projectBReferences = net80Refs
            .Concat(ReferencesHelper.GetReferences(ReferencesSet.MessagePack))
            .Append(projectARef);

        CSharpCompilation projectBCompilation = CSharpCompilation.Create(
            "ProjectB",
            new[] { CSharpSyntaxTree.ParseText(projectBSource, new CSharpParseOptions(LanguageVersion.Latest)) },
            projectBReferences,
            new CSharpCompilationOptions(OutputKind.DynamicallyLinkedLibrary));

        GeneratorDriver driver = CSharpGeneratorDriver.Create(new MessagePackGenerator().AsSourceGenerator());
        driver = driver.RunGeneratorsAndUpdateCompilation(projectBCompilation, out Compilation outputCompilation, out ImmutableArray<Diagnostic> generatorDiagnostics);
        GeneratorDriverRunResult runResult = driver.GetRunResult();

        generatorDiagnostics.Where(d => d.Severity == DiagnosticSeverity.Error).Should().BeEmpty();

        var generatedCode = runResult.Results
            .SelectMany(r => r.GeneratedSources)
            .Select(s => s.SourceText.ToString())
            .FirstOrDefault(text => text.Contains("ExternalTypeFormatter"));

        generatedCode.Should().NotBeNull("source generator should produce a partial class implementation for ExternalTypeFormatter");
        generatedCode.Should().Contain("Serialize");
        generatedCode.Should().Contain("Deserialize");

        var compilationErrors = outputCompilation.GetDiagnostics()
            .Where(d => d.Severity == DiagnosticSeverity.Error)
            .ToList();
        compilationErrors.Should().BeEmpty("output compilation should have no errors after generator runs");

        using var projectBStream = new MemoryStream();
        var emitResult = outputCompilation.Emit(projectBStream);
        emitResult.Success.Should().BeTrue("output compilation should emit successfully");
    }
}
